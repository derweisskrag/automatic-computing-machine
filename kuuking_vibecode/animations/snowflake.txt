use macroquad::prelude::*;

struct Snowflake {
    x: f32,
    y: f32,
    radius: f32,
    speed: f32,
    drift: f32,
    phase: f32,
}

impl Snowflake {
    fn new(width: f32, height: f32) -> Self {
        Self {
            x: rand::gen_range(0.0, width),
            y: rand::gen_range(-height, height),
            radius: rand::gen_range(2.0, 5.0),
            speed: rand::gen_range(12.0, 34.0),
            drift: rand::gen_range(-18.0, 18.0),
            phase: rand::gen_range(0.0, std::f32::consts::TAU),
        }
    }

    fn reset(&mut self, width: f32, height: f32) {
        self.x = rand::gen_range(0.0, width);
        self.y = rand::gen_range(-height * 0.2, -10.0);
        self.radius = rand::gen_range(2.0, 5.0);
        self.speed = rand::gen_range(12.0, 34.0);
        self.drift = rand::gen_range(-18.0, 18.0);
        self.phase = rand::gen_range(0.0, std::f32::consts::TAU);
    }

    fn update(&mut self, width: f32, height: f32) {
        let dt = get_frame_time() * 60.0;
        self.y += self.speed * dt;
        self.x += self.drift * dt * 0.6 + self.phase.sin() * 0.35;

        if self.y > height + self.radius {
            self.reset(width, height);
        }

        if self.x < -self.radius {
            self.x = width + self.radius;
        } else if self.x > width + self.radius {
            self.x = -self.radius;
        }
    }

    fn draw(&self) {
        let alpha = 0.75;
        let color = Color::new(0.9, 0.96, 1.0, alpha);
        draw_circle(self.x, self.y, self.radius, color);

        let sparkle = self.radius * 1.6;
        draw_circle(self.x - self.radius * 0.6, self.y - self.radius * 0.6, self.radius * 0.45, Color::new(1.0, 1.0, 1.0, 0.5));
        draw_circle(self.x + sparkle * 0.2, self.y + sparkle * 0.2, self.radius * 0.35, Color::new(1.0, 1.0, 1.0, 0.35));
    }
}

#[macroquad::main("Snowflakes")]
async fn main() {
    let mut flakes = Vec::new();
    for _ in 0..220 {
        let width = screen_width();
        let height = screen_height();
        flakes.push(Snowflake::new(width, height));
    }

    loop {
        let width = screen_width();
        let height = screen_height();

        clear_background(Color::new(0.04, 0.07, 0.12, 1.0));

        for flake in &mut flakes {
            flake.update(width, height);
            flake.draw();
        }

        next_frame().await;
    }
}
